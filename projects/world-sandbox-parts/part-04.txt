      const arr=cTiles[cid];
      for(let yy=0;yy<h;yy++)for(let xx=0;xx<w;xx++){
        const si=(yy*w+xx)*4;if(img[si]<=128)continue;
        const tx=x0+xx,ty=y0+yy,ti=ty*WORLD_W+tx;
        if(legacyCountry[ti]===-1){legacyCountry[ti]=cid;arr.push(ti);}
      }
    }
    if(isMobile||(cid&7)===0){setLoad(18+Math.round(cid/Math.max(1,legacyFeats.length)*22),'Restoring original TNO geography…');await new Promise(r=>setTimeout(r,0));}
  }

  let oldPid=0;
  const oldCenters=[]; // {pid,cid,cx,cy}; pid order MUST match original snapshot arrays
  for(let cid=0;cid<cTiles.length;cid++){
    const tiles=cTiles[cid];if(!tiles.length)continue;
    // The per-country raster loop itself is y-major then x-major, so these tile
    // indices are already strictly ascending. Sorting them again (especially
    // Russia-sized arrays) was pure O(N log N) TNO startup work with no output change.
    let minX=WORLD_W,maxX=0,minY=WORLD_H,maxY=0;
    for(const ti of tiles){const x=ti%WORLD_W,y=(ti/WORLD_W)|0;if(x<minX)minX=x;if(x>maxX)maxX=x;if(y<minY)minY=y;if(y>maxY)maxY=y;}
    const name=legacyFeats[cid]?.properties?.name||String(cid);
    const histPtCount=(HIST_PROVINCES[name]?.length)||0;
    const k=Math.max(1,Math.max(
      Math.round(tiles.length/VORONOI_TARGET),
      Math.round(Math.sqrt((maxX-minX)**2+(maxY-minY)**2)/28),
      histPtCount
    ));
    const rng=makeRng(hashStr(name)),sample=tiles.length>800?sampleArr(tiles,800):tiles,seeds=[];
    seeds.push(tiles[Math.floor(rng()*tiles.length)]);
    // Incremental k-means++ distances are mathematically identical to rescanning all
    // prior seeds, but reduce seed selection from O(sample×k²) to O(sample×k).
    const minD=new Float64Array(sample.length);minD.fill(Infinity);
    const updateMin=st=>{const sx0=st%WORLD_W,sy0=(st/WORLD_W)|0;for(let i=0;i<sample.length;i++){const t=sample[i],x=t%WORLD_W,y=(t/WORLD_W)|0,d=(x-sx0)*(x-sx0)+(y-sy0)*(y-sy0);if(d<minD[i])minD[i]=d;}};
    updateMin(seeds[0]);
    while(seeds.length<k){
      let total=0,comp=0;for(let i=0;i<minD.length;i++){const y=minD[i]-comp,t=total+y;comp=(t-total)-y;total=t;}
      let r=rng()*total,chosen=sample[sample.length-1];for(let i=0;i<sample.length;i++){r-=minD[i];if(r<=0){chosen=sample[i];break;}}
      seeds.push(chosen);updateMin(chosen);
    }
    const sx=seeds.map(t=>t%WORLD_W),sy=seeds.map(t=>(t/WORLD_W)|0);
    _legacyTnoLloydRelax(sx,sy,tiles,5,cid,legacyCountry,minX,maxX,minY,maxY);
    const nearest=_legacyTnoSeedNearestFactory(sx,sy),cnt=new Int32Array(k),sumX=new Float64Array(k),sumY=new Float64Array(k);
    for(let tii=0;tii<tiles.length;tii++){
      const ti=tiles[tii],x=ti%WORLD_W,y=(ti/WORLD_W)|0,best=nearest(x,y);cnt[best]++;sumX[best]+=x;sumY[best]+=y;
    }
    const seedPid=new Int32Array(k);seedPid.fill(-1);
    for(let gi=0;gi<k;gi++){
      if(cnt[gi]<3)continue;const pid=oldPid++;seedPid[gi]=pid;oldCenters.push({pid,cid,cx:sumX[gi]/cnt[gi],cy:sumY[gi]/cnt[gi]});
    }
    // Exact equivalent of legacyTileProv[x,y], evaluated only for sample points.
    legacySeedMaps[cid]={nearest,seedPid};
    if(isMobile||(cid&3)===0){setLoad(40+Math.round(cid/Math.max(1,cTiles.length)*40),'Rebuilding original TNO provinces…');await new Promise(r=>setTimeout(r,0));}
  }

  if(oldPid!==_TNO_LEGACY_LAND_COUNT){
    throw new Error(`TNO legacy reconstruction mismatch: expected ${_TNO_LEGACY_LAND_COUNT} original land provinces, rebuilt ${oldPid}. TNO was NOT applied.`);
  }

  const byCountry=Array.from({length:legacyFeats.length},()=>[]);
  for(const c of oldCenters)byCountry[c.cid].push(c);
  const legacyPidAt=(cid,x,y)=>{
    if(x<0||y<0||x>=WORLD_W||y>=WORLD_H)return -1;
    const ti=y*WORLD_W+x;if(legacyCountry[ti]!==cid)return -1;
    const sm=legacySeedMaps[cid];if(!sm)return -1;
    return sm.seedPid[sm.nearest(x,y)]??-1;
  };
  const remap=new Int32Array(provinces.length);remap.fill(-1);
  // At most 13 overlap samples are used per current province. Reuse tiny typed
  // vote buffers instead of allocating thousands of Map objects during TNO load.
  const voteIds=new Int16Array(13),voteCounts=new Uint8Array(13);
  let mapped=0;
  for(let pid=0;pid<provinces.length;pid++){
    const p=provinces[pid];if(!p||p.isSea||p.countryId<0||p.countryId>=_tnoWorldToLegacyCountryId.length)continue;
    const legacyCid=_tnoWorldToLegacyCountryId[p.countryId];if(legacyCid<0)continue;
    const ll=proj.invert([p.centerTx,p.centerTy]);if(!ll)continue;
    const q=legacyProj(ll);if(!q)continue;
    const qx=Math.round(q[0]),qy=Math.round(q[1]);
    if(qx<bounds.x0||qx>bounds.x1||qy<bounds.y0||qy>bounds.y1)continue;

    // Majority-overlap transfer: sample across the new province rather than trusting
    // one centroid. This preserves original TNO borders on the new province mesh much
    // more faithfully around narrow fronts / fragmented warlord states.
    const pts=p.tiles||[],sampleN=Math.min(13,pts.length);let voteLen=0;
    if(sampleN){
      for(let si=0;si<sampleN;si++){
        const ti=pts[Math.min(pts.length-1,Math.floor((si+.5)*pts.length/sampleN))];
        const tx=ti%WORLD_W,ty=(ti/WORLD_W)|0,ll2=proj.invert([tx,ty]);if(!ll2)continue;
        const q2=legacyProj(ll2);if(!q2)continue;const x=Math.round(q2[0]),y=Math.round(q2[1]);
        if(x<bounds.x0||x>bounds.x1||y<bounds.y0||y>bounds.y1)continue;
        const op=legacyPidAt(legacyCid,x,y);if(op<0)continue;
        let vi=0;for(;vi<voteLen;vi++)if(voteIds[vi]===op)break;
        if(vi===voteLen){voteIds[voteLen]=op;voteCounts[voteLen]=1;voteLen++;}else voteCounts[vi]++;
      }
    }
    let old=-1,bestVotes=-1;
    for(let vi=0;vi<voteLen;vi++){const n=voteCounts[vi],op=voteIds[vi];if(n>bestVotes){bestVotes=n;old=op;}}

    // Centre tile breaks ties / handles very small provinces.
    if(qx>=0&&qx<WORLD_W&&qy>=0&&qy<WORLD_H){
      const op=legacyPidAt(legacyCid,qx,qy);
      if(op>=0){let n=0;for(let vi=0;vi<voteLen;vi++)if(voteIds[vi]===op){n=voteCounts[vi];break;}if(n>=bestVotes){old=op;bestVotes=n;}}
    }
    for(let vi=0;vi<voteLen;vi++)voteCounts[vi]=0;
    // Coast/rounding fallback: nearest original province CENTRE in the same modern country.
    if(old<0){
      const arr=byCountry[legacyCid];let bd=Infinity;
      for(const c of arr){const d=(c.cx-q[0])**2+(c.cy-q[1])**2;if(d<bd){bd=d;old=c.pid;}}
      if(bd>30*30)old=-1; // never drag overseas/out-of-footprint provinces into TNO
    }
    if(old>=0){remap[pid]=old;mapped++;}
  }
  if(mapped<Math.min(500,provinces.length*0.03))throw new Error(`TNO legacy remap mapped only ${mapped} full-world provinces; refusing to corrupt the scenario.`);
  _tnoLegacyProvinceRemap=remap;_storeTnoRemap(remap);
  // Keep source geography resident so a stale/corrupt cache can be rebuilt safely in-session.
  setLoad(82,`TNO geography restored (${mapped} full-world provinces mapped)…`);
  return remap;
}

async function _ensureTnoLegacyProvinceRemap(){
  if(_tnoLegacyProvinceRemap&&_tnoLegacyProvinceRemap.length===provinces.length)return _tnoLegacyProvinceRemap;
  if(!_tnoLegacyRemapPromise)_tnoLegacyRemapPromise=_buildTnoLegacyProvinceRemap().catch(err=>{_tnoLegacyRemapPromise=null;throw err;});
  return _tnoLegacyRemapPromise;
}

function _prepareTnoCivIdMap(){
  // Original TNO IDs 75..111 were dynamic states in the regional build.
  // The full-world build has ordinary countries in those numeric slots, so NEVER
  // address TNO nations by their old array index directly. Base states are located
  // by stable legacy countryId; custom TNO states are appended after all world civs.
  let maxOld=0;for(const cd of TNO_CIV_TEMPLATE)if(cd.id>maxOld)maxOld=cd.id;
  const map=new Int32Array(maxOld+1);map.fill(-1);

  // Restore base legacy nations by country identity, not compact civ-array position.
  for(const cd of TNO_CIV_TEMPLATE){
    if(cd.customNation)continue;
    const worldCountryId=(_tnoLegacyToWorldCountryId&&cd.countryId>=0&&cd.countryId<_tnoLegacyToWorldCountryId.length)?_tnoLegacyToWorldCountryId[cd.countryId]:-1;
    let civ=worldCountryId>=0?civs.find(c=>c&&!c.customNation&&c.countryId===worldCountryId):null;
    if(!civ)civ=civs.find(c=>c&&!c.customNation&&c.name===cd.name);
    if(!civ)throw new Error(`TNO base nation missing from full-world map: ${cd.name} (legacy country ${cd.countryId}, world country ${worldCountryId})`);
    map[cd.id]=civ.id;
    if(cd.name)civ.name=cd.name;
    if(cd.color&&/^#[0-9a-f]{6}$/i.test(cd.color)){
      civ.color=cd.color;civ.terrAbgr=hexToAbgr(lighten(cd.color,.20));civ.borderAbgr=hexToAbgr(lighten(cd.color,.35));
      civ._cd={base:cd.color,baseD:darken(cd.color,.22),helm:darken(cd.color,.38),helmD:darken(darken(cd.color,.38),.12),lite:lighten(cd.color,.50)};
    }
  }

  // Append custom TNO states so they can never overwrite newly-added world countries.
  for(const cd of TNO_CIV_TEMPLATE){
    if(!cd.customNation)continue;
    const civ=_makeCustomCiv(cd.name,cd.color,-1);
    civ.countryId=cd.countryId;
    civ.customNation=true;
    civ._tnoLegacyId=cd.id;
    map[cd.id]=civ.id;
  }
  _tnoCivIdMap=map;
  return map;
}

  return oldId>=0&&_tnoCivIdMap&&oldId<_tnoCivIdMap.length?_tnoCivIdMap[oldId]:-1;
}
function _validateTnoApplyInputs(snap,exactSnapshot){
  const validIds=new Set(TNO_CIV_TEMPLATE.map(c=>c.id));
  if(!snap||!Array.isArray(snap.owners)||!snap.owners.length)throw new Error('TNO snapshot has no ownership data.');
  if(!exactSnapshot){
    // Legacy snapshots contain 1,469 LAND ownership slots followed by 43 explicit
    // -1 non-land placeholders. The remap indexes land slots only, so requiring the
    // whole serialized array to equal 1,469 incorrectly rejected every valid TNO map.
    if(snap.owners.length<_TNO_LEGACY_LAND_COUNT)throw new Error(`TNO snapshot province count mismatch: expected at least ${_TNO_LEGACY_LAND_COUNT} land slots, got ${snap.owners.length}.`);
    for(let i=_TNO_LEGACY_LAND_COUNT;i<snap.owners.length;i++)if(snap.owners[i]!==-1)throw new Error(`TNO snapshot has unexpected land ownership beyond legacy land slot ${_TNO_LEGACY_LAND_COUNT-1} (index ${i}).`);
    const r=_tnoLegacyProvinceRemap;if(!r||r.length!==provinces.length)throw new Error('Original TNO province geography has not been reconstructed yet.');
    let mapped=0;for(let i=0;i<r.length;i++){const v=r[i];if(v>=0){if(v>=snap.owners.length)throw new Error(`TNO remap points outside snapshot at current province ${i}.`);mapped++;}}
    if(mapped<Math.min(500,provinces.length*0.03))throw new Error(`TNO remap coverage is unsafe (${mapped} mapped provinces).`);
  }else if(snap.owners.length!==provinces.length) throw new Error('Exact TNO snapshot does not match the current province mesh.');
  const used=new Set(snap.owners.filter(v=>Number.isInteger(v)&&v>=0));for(const id of used)if(!validIds.has(id))throw new Error(`TNO snapshot references unknown legacy nation id ${id}.`);
  if(!_ensureTnoCountryIdMaps())throw new Error('TNO country compatibility map is unavailable.');
  // Verify every non-custom template state can be located BEFORE _prepareTnoCivIdMap()
  // changes a single live country name/color. This makes failure truly transactional.
  for(const cd of TNO_CIV_TEMPLATE){if(cd.customNation)continue;const worldCountryId=cd.countryId>=0&&cd.countryId<_tnoLegacyToWorldCountryId.length?_tnoLegacyToWorldCountryId[cd.countryId]:-1;const found=(worldCountryId>=0&&civs.some(c=>c&&!c.customNation&&c.countryId===worldCountryId))||civs.some(c=>c&&!c.customNation&&c.name===cd.name);if(!found)throw new Error(`TNO base nation preflight failed: ${cd.name}.`);}
  return true;
}

function _applyBuiltInTnoMap(scenId){
  const snap=TNO_SCENARIO_DATA[scenId];
  if(!snap) return false;
  if(!Array.isArray(snap.owners) || !snap.owners.length) return false;
  // Identify the original regional TNO format by structure, not by an accidental
  // array-length comparison with the generated full-world province mesh.
  let legacySnapshot=snap.owners.length>=_TNO_LEGACY_LAND_COUNT;
  if(legacySnapshot)for(let i=_TNO_LEGACY_LAND_COUNT;i<snap.owners.length;i++)if(snap.owners[i]!==-1){legacySnapshot=false;break;}
  const exactSnapshot=!legacySnapshot&&snap.owners.length===provinces.length;

  // PRE-FLIGHT EVERYTHING before mutating names/colors or appending custom civs.
  // Previous builds prepared TNO civs first and only then discovered a bad/missing remap,
  // leaving the live world half-converted after a failed load.
  _validateTnoApplyInputs(snap,exactSnapshot);
  const civMap=_prepareTnoCivIdMap();
  // Verify every owner used by this snapshot resolves to a live civ before the first province write.
  for(const legacyId of new Set(snap.owners)){if(legacyId>=0&&(legacyId>=civMap.length||civMap[legacyId]<0||!civs[civMap[legacyId]]))throw new Error(`TNO nation mapping failed for legacy owner ${legacyId}.`);}

  // Full-world TNO rule: ONLY the footprint represented by the original TNO map
  // is replaced. Provinces outside that old regional map remain on the full-world
  // base map instead of being stretched into unrelated TNO owners.
  const writeOwner=(pid,legacyOwnerId)=>{
    if(legacyOwnerId<0||!provinces[pid])return;
    const civId=legacyOwnerId<civMap.length?civMap[legacyOwnerId]:-1;
    if(civId<0||civId>=civs.length)return;
    const old=provOwner[pid];
    if(old===civId)return;
    if(old>=0&&civs[old]){civs[old].ownedProvs.delete(pid);civs[old].territory.deleteProvince?.(pid);}
    provOwner[pid]=civId;civs[civId].ownedProvs.add(pid);civs[civId].territory.addProvince?.(pid);bakeProvOwnedColor(pid,civId);
  };

  if(exactSnapshot){
    _fillBaseWorldColors(ownedColorBuf);tileCivBuf.fill(-1);_provDivIndexDirty=true;_provDivIndex.clear();
    for(const civ of civs){civ.ownedProvs.clear();civ.territory.resetProvinces?.([]);_clearCivRenderMovers(civ);civ.foxes.length=0;civ._damagedDivs?.clear();civ.isPlayer=false;}
    provOwner.fill(-1);
    for(let pid=0;pid<snap.owners.length;pid++)writeOwner(pid,snap.owners[pid]);
  }else{
    for(let pid=0;pid<provinces.length;pid++){
      const oldPid=_tnoLegacyProvinceRemap[pid];
      if(oldPid<0||oldPid>=_TNO_LEGACY_LAND_COUNT)continue;
      const owner=snap.owners[oldPid];
      if(owner>=0)writeOwner(pid,owner);
    }
  }

  const defaultTechIds=_buildTechSet(3,3);
  for(const civ of civs){
    civ.res=civStartRes(civ.ownedProvs.size);civ.techs=new Set(defaultTechIds);_recomputeBotTechBonus(civ);civ._techCooldown=civ.id%60;
  }

  for(const civ of civs)civ._territoryVersion=((civ._territoryVersion||1)+1)>>>0||1;
  _touchRouteTopology();_botWarCacheDirty=true;_markAllBotStrategyDirty();
  _terrBorderDirty=true;_allianceBorderDirty=true;_allianceLabelDirty=true;_alOverlayDirty=true;_civLabelDirty=true;
  _civOutlineCache.clear();_civOutlinePathCache.clear();_clearLabelImgCache();_clearFoxSpriteCache();_markWorldFullDirty();_hlDirty=true;_markCapitalsDirty();
  // Keep raw source geography available. A localStorage remap cache can be cleared,
  // rejected or version-invalidated later in the same session; discarding this data made
  // the next TNO load impossible to repair. The heavy raster reconstruction buffers are
  // still local to _buildTnoLegacyProvinceRemap() and are released normally.
  return true;
}

function _tnoNativeNearestSeed(lon,lat,seeds){
  let best='',bestD=Infinity;
  for(const [name,pt] of Object.entries(seeds)){
    const dx=(lon-pt[0])*Math.cos((lat+pt[1])*Math.PI/360),dy=lat-pt[1],d=dx*dx+dy*dy;
    if(d<bestD){bestD=d;best=name;}
  }
  return best;
}
function _tnoNative1962Owner(base,lon,lat){
  base=String(base||'');

  if(base==='Russia'){

    if(lon<43.2 && lat<61.8) return 'Reichskommissariat Moskowien';
    if(lat>69 || (lon>100&&lat>66)) return 'Unorganized Siberian Territories';
    return _tnoNativeNearestSeed(lon,lat,TNO_NATIVE_1962_RUSSIA_SEEDS)||'Unorganized Siberian Territories';
  }
  if(base==='China'){
    if(lon<91) return 'Xinjiang Border Defense Administration';
    if(lon<103 && lat<35 && lat>27) return 'Kingdom of Tibet';

    if(lon>108 && lat<25.5) return 'State of Guangdong';
    if(lon<106 && lat<30.5) return 'NRA 40th Army Group';
    if(lon<110 && lat<34.5) return 'NRA 24th Army';
    return 'Republic of China';
  }
  if(base==='Mongolia') return lon>108?'Empire of Manchuria':'Republic of China';
  if(base==='India'){
    if(lon>87.0 || (lon>84.5&&lat<25.5)) return 'Government of Free India';
    return 'Republic of India';
  }
  if(base==='Bolivia'){
    // Santa Cruz de la Sierra and the Paraguayan-backed eastern commission.
    if(lon>-65.2 && lat<-14.0) return 'Oriental Government Commission';
    return 'Republic of Bolivia';
  }
  if(base==='Uzbekistan'){
    // Turkestan Legion is based around Qarshi in southern Uzbekistan.
    if(lat<40.4 && lon>63.5) return 'Turkestan Legion';
    return 'Uzbek Soviet Socialist Republic';
  }

    if(lon<-75) return 'Chilean Antarctic Territory';

// Apply a scenario: reassign province ownership according to scenario.map.
// Called once after initCivs(), before the game starts.
function applyScenario(scenId){
  const scen = SCENARIOS[scenId];
  if(!scen) return;
  if(scen.unstableSmp){_applyUnstableSmpMap();return;}
  if(scen.nativeTno1962){_applyNativeTno1962Map();return;}
  if(scen.historicalKey==='ww1_1914'){_applyWW11914Map();return;}
  if(scen.historicalKey==='ww1_1917'){_applyWW11917Map();return;}
  if(scen.historicalYear){_applyInterwarWW2Map(scen.historicalYear);return;}
  if(scen.snapshotKey){
    _applyBuiltInTnoMap(scen.snapshotKey);
    return;
  }
  if(!scen.map || Object.keys(scen.map).length === 0) return;

  // Build name→civId lookup
  const civByName = {};
  for(const civ of civs) civByName[civ.name] = civ.id;

  for(const [targetName, sourcesArr] of Object.entries(scen.map)){
    const targetId = civByName[targetName];
    if(targetId === undefined) continue;

    for(const sourceName of sourcesArr){
      const sourceId = civByName[sourceName];
      if(sourceId === undefined || sourceId === targetId) continue;
      // Transfer all provinces from source → target
      const srcProvs = Array.from(civs[sourceId].ownedProvs);
      for(const pid of srcProvs){
        civClaimProvince(pid, targetId);
      }
    }
  }

  // Apply colour overrides
  if(scen.colors){
    for(const [name, col] of Object.entries(scen.colors)){
      const cid = civByName[name];
      if(cid === undefined) continue;
      civs[cid].color = col;
      civs[cid].terrAbgr = hexToAbgr(lighten(col, .20));
      civs[cid].borderAbgr = hexToAbgr(lighten(col, .35));
      // Rebake all owned province colours
      for(const pid of civs[cid].ownedProvs) bakeProvOwnedColor(pid, cid);
    }
    _clearFoxSpriteCache(); // colours changed — flush cached sprites
  }

  // Apply historical name overrides (e.g. "Russia" → "Soviet Union")
  if(scen.names){
    for(const [modernName, historicalName] of Object.entries(scen.names)){
      const cid = civByName[modernName];
      if(cid !== undefined) civs[cid].name = historicalName;
    }
  }
  // Labels need to be rebuilt after name changes
  _civLabelDirty = true;
  _clearLabelImgCache();
}

let _menuLandBounds=null;
function _getMenuLandBounds(){
  if(_menuLandBounds) return _menuLandBounds;
  if(!_provRenderMinX||_provRenderMinX.length!==provinces.length) _buildProvinceRenderBounds();
  let minX=WORLD_W,maxX=0,minY=WORLD_H,maxY=0,found=false;
  for(let pid=0;pid<provinces.length;pid++){
    const p=provinces[pid];
    if(!p||p.isSea||!p.tiles?.length) continue;
    found=true;
    if(_provRenderMinX[pid]<minX)minX=_provRenderMinX[pid];
    if(_provRenderMaxX[pid]>maxX)maxX=_provRenderMaxX[pid];
    if(_provRenderMinY[pid]<minY)minY=_provRenderMinY[pid];
    if(_provRenderMaxY[pid]>maxY)maxY=_provRenderMaxY[pid];
  }
  if(!found) return (_menuLandBounds={minX:0,maxX:WORLD_W,minY:0,maxY:WORLD_H});
  const pad=12;
  return (_menuLandBounds={
    minX:Math.max(0,minX-pad),maxX:Math.min(WORLD_W,maxX+pad),
    minY:Math.max(0,minY-pad),maxY:Math.min(WORLD_H,maxY+pad)
  });
}
function centerMenuWorldMap(){
  if(!canvas || !mapReady) return;
  const b=_getMenuLandBounds();
  const spanX=Math.max(1,b.maxX-b.minX),spanY=Math.max(1,b.maxY-b.minY);
  // COVER: map fills the entire start screen. Small overscan prevents seams.
  const fitX=canvas.width /(spanX*TS);
  const fitY=canvas.height/(spanY*TS);
  zoom=Math.max(0.04,Math.min(0.34,Math.max(fitX,fitY)*1.025));
  const midX=(b.minX+b.maxX)*0.5,midY=(b.minY+b.maxY)*0.5;
  camX=midX-canvas.width /(2*TS*zoom);
  camY=midY-canvas.height/(2*TS*zoom);
  clampCam();
  _menuMapFrameValid=false;
}

function buildCountryBar(filter=''){
  const list = document.getElementById('cb-list');
  const fl = filter.toLowerCase();
  const sorted = [...civs].filter(c => c.ownedProvs.size > 0).sort((a,b)=>a.name.localeCompare(b.name));
  list.innerHTML = '';
  for(const civ of sorted){
    if(fl && !civ.name.toLowerCase().includes(fl)) continue;
    const item = document.createElement('div');
    item.className = 'cb-item' + (_selectedCbCivId === civ.id ? ' active' : '');
    item.innerHTML = `<span class="cb-dot" style="background:${civ.color}"></span><span class="cb-name">${civ.name}</span><span class="cb-provs">${civ.ownedProvs.size}p</span>`;
    item.addEventListener('click', ()=>{
      _selectedCbCivId = civ.id;
      document.querySelectorAll('.cb-item').forEach(i=>i.classList.remove('active'));
      item.classList.add('active');
      document.getElementById('cb-play').disabled = false;
      document.getElementById('cb-chosen').textContent = civ.name + ' selected';
      // Highlight this civ on the map
      panelCivId = civ.id; _hlDirty = true;
      // Pan camera to capital
      const capPid=_capitalProvinceForCiv(civ.id);
      if(capPid>=0){

        if(cap){
          camX = cap.centerTx - canvas.width/(2*TS*zoom);
          camY = cap.centerTy - canvas.height/(2*TS*zoom);
          clampCam();
        }
      }
    });
    list.appendChild(item);
  }
}

// Start the game WITHOUT wiping the current map state.
// Used by both the editor path and the save-load path.
function _startGamePreserveMap(civId, mode){
  gameMode     = mode;
  playerCivId  = civId;

  // Clear any stale isPlayer flag then set correct one
  for(const civ of civs) civ.isPlayer = false;
  if(civId >= 0) civs[civId].isPlayer = true;

  // Camera: centre on the chosen civ's first province
  if(civId >= 0 && civs[civId].ownedProvs && civs[civId].ownedProvs.size > 0){
    const pid = [...civs[civId].ownedProvs][0];
    const cap = provinces[pid];
    if(cap){
      zoom = 0.30;
      camX = cap.centerTx - canvas.width  / (2*TS*zoom);
      camY = cap.centerTy - canvas.height / (2*TS*zoom);
      clampCam();
    }
  }

  // HUD
  if(civId >= 0){
    document.getElementById('hud-dot').style.background = civs[civId].color;
    document.getElementById('hud-civname').textContent  = civs[civId].name.toUpperCase();
    document.getElementById('hud-mode').textContent     = SCENARIOS[activeScenario]?.label || '';
  } else {
    const label = SCENARIOS[activeScenario]?.label || '';
    document.getElementById('hud-dot').style.background  = 'transparent';
    document.getElementById('hud-civname').textContent   = 'WORLD SANDBOX';
    document.getElementById('hud-mode').textContent      = mode==='spectator'
      ? `SPECTATING${label?' · '+label:''}` : mode==='sandbox' ? 'SANDBOX' : '';
  }

  hideCountryBar();
  panelCivId = -1; selectedProv = -1; _hlDirty = true;
  document.getElementById('top-bar').classList.add('game-visible');
  document.getElementById('gen-bar').classList.add('game-visible');

  // Load path: promote the selected country's saved technology set into the
  // player tree BEFORE any UI/bonus cache is rebuilt. This fixes the old
  // main-menu load path that restored the world but silently reset your tech tree.
  if(_countryBarMode === 'load' && civId >= 0){
    playerTechs.clear();
    for(const id of _filterKnownTechIds([...(civs[civId].techs || [])])) playerTechs.add(id);
    _recomputeTechBonuses();
    _syncPlayerTechsToCiv();
    _divCapCache = -1;
    _clearFoxSpriteCache();
  }

  // Editor path: apply alliances & techs as a fresh start
  if(_countryBarMode === 'editor'){
    if(startingAlliancesEnabled) applyStartingAlliances(activeScenario);
    const playerCivName = civId >= 0 ? (civs[civId]?.name || '') : '';
    applyStartingTechs(activeScenario, playerCivName);
    const scen = SCENARIOS[activeScenario];
    if(scen && scen.startDate){
      gameDay=scen.startDate.day; gameMonth=scen.startDate.month; gameYear=scen.startDate.year;
    } else { gameDay=1; gameMonth=1; gameYear=2024; }
    gameDayFrac = 0;
    initBotPhases();
    for(const civ of civs){
      if(civ.id === playerCivId) continue;
      if(!civ.techs || !civ.techs.size){
        if(activeScenario==='ww1_1914') civ.techs=new Set(_ww1TechForName(civ.name));
        else {
          const lv = MODERN_NATION_TECHS[civ.name];
          civ.techs = new Set(_buildTechSet(lv?lv[0]:3, lv?lv[1]:3));
        }
      }
      _recomputeBotTechBonus(civ);
      if(!SCENARIOS[activeScenario]?.historicalYear&&activeScenario!=='ww1_1914'&&activeScenario!=='ww1_1917'&&!_isTnoScenario(activeScenario)){
        const lv = MODERN_NATION_TECHS[civ.name];
        const tier = lv ? Math.max(lv[0],lv[1]) : 3;
        civ.res.coins = Math.max(civ.res.coins, 400+tier*250);
      }
      civ._techCooldown = civ.id % 60;
    }
  }
  // Editor maps define a new starting world, so province/city home ownership must
  // become the new baseline. Loaded saves already restored their saved baselines.
  if(_countryBarMode === 'editor'){
    _captureOriginalMapOwners(true);
    _captureCityOriginalOwners(true);
  } else if(_countryBarMode === 'load'){
    if(!_originalMapOwnerIds) _restoreOriginalMapOwners(null);
    if(!_cityOriginalOwnerIds) _restoreCityOriginalOwners(null);
  }
  // Load path: techs/alliances/date already fully restored — nothing extra needed

  _countryBarMode = 'normal'; // reset for next time
  _normalizeFreshGameplayRenderState();
  gameReady = true;
  document.body.classList.add('game-running');
  if(playerAutoMoveUniversal)for(const d of playerDivs())d.autoMove=true;
  setGameSpeed(0);
  setTimeout(()=>{ buildGenBar(); updateFlagBanner(); buildAllianceWidget(); if(playerAutoMoveUniversal)_refreshUniversalAutoMovePlan(); else updateMilBar(); _tryStartQueuedPeaceConferences(); }, 50);
}

// Restore save-file world state (ownership, buildings, alliances, bots, date)
// WITHOUT touching playerCivId/gameReady/top-bar — caller shows country bar next.
function _applyLoadedSaveState(text){
  const jsonStr = text.split('\n').filter(l=>!l.startsWith('//')).join('\n');
  const state   = JSON.parse(jsonStr);
  if(!state || state.version !== 1) throw new Error('Invalid or unsupported save format');
  if(!mapReady) throw new Error('Map not loaded yet — wait for loading to finish');

  _fullReset();                               // wipe previous game
  activeScenario = state.scenario || 'modern';
  gameMode       = state.gameMode  || 'sandbox';
  playerAutoMoveUniversal=!!state.playerAutoMoveUniversal;
  if(state.botDifficulty&&BOT_DIFFICULTY_PROFILES[state.botDifficulty]){botDifficulty=state.botDifficulty;const bd=document.getElementById('set-bot-difficulty');if(bd)bd.value=botDifficulty;}

  // Recreate released/custom nations before ownerArr references their ids.
  // This load path is used by the main-menu "Load Save" flow (before the
  // player chooses a nation), so it must mirror loadGameFromText().
  for(const cd of [...(state.civData||[])].sort((a,b)=>(a.id??0)-(b.id??0))){
    if((cd.id??-1)>=civs.length) _restoreDynamicCivFromSave(cd);
  }
  for(const cd of (state.civData||[])) if(cd.id>=0&&cd.id<civs.length) _restoreDynamicCivFromSave(cd);

  // Restore province ownership
  const saved = state.ownerArr;
  if(saved && saved.length === provOwner.length){
    for(let pid=0; pid<provinces.length; pid++){
      if(provOwner[pid] >= 0) restoreProvUnownedColor(pid);
    }
    for(const civ of civs){ civ.ownedProvs.clear(); civ.territory.resetProvinces?.([]); }
    provOwner.fill(-1);
    for(let pid=0; pid<saved.length; pid++){
      const civId = saved[pid];
      if(civId >= 0 && civId < civs.length){
        provOwner[pid] = civId;
        civs[civId].ownedProvs.add(pid);
        bakeProvOwnedColor(pid, civId);
      }
    }
  }

  _restoreOriginalMapOwners(state.originalMapOwners);
  _restoreCityOriginalOwners(state.cityOriginalOwners);
  _destroyedCitySourceIds.clear();_markCapitalsDirty(); // migrate legacy saves: all city markers are restored permanently

  // Restore civ resources and technology sets. This path runs before the user
  // chooses which nation to play, so tech progress must live on each civ until
  // _startGamePreserveMap() promotes the chosen country's set into playerTechs.
  const _savedPlayerTechs = _deserializeSavedPlayerTechs(state);
  const _civsWithSavedTechState = new Set();
  for(const cd of (state.civData||[])){
    if(cd.id < 0 || cd.id >= civs.length) continue;
    if(cd.res) Object.assign(civs[cd.id].res, cd.res);
    const hasPlayerTree = cd.id === state.playerCivId &&
      ((state.techTreeData && (Array.isArray(state.techTreeData.econ) || Array.isArray(state.techTreeData.mil))) || Array.isArray(state.techData));
    const hasExplicitTechs = Array.isArray(cd.techs) || hasPlayerTree || Array.isArray(cd.botTechs);
    const savedTechs = Array.isArray(cd.techs) ? cd.techs
      : (hasPlayerTree ? _savedPlayerTechs
      : (Array.isArray(cd.botTechs) ? cd.botTechs : []));
    civs[cd.id].techs = new Set(_filterKnownTechIds(savedTechs));
    if(hasExplicitTechs) _civsWithSavedTechState.add(cd.id);

  // Restore wars + peace-conference metadata/treaties.

  }
  // Backward-compatible old saves: synthesize snapshots from the restored war.
  for(const key of warPairs){const [a,b]=_decodeWarKey(key);if(!warRecords.has(key))_ensureWarRecord(a,b);}

  // Restore province buildings
  if(state.provBld && state.provBld.length === provinces.length){
    for(let pid=0; pid<provinces.length; pid++)
      Object.assign(provinces[pid].bld, state.provBld[pid]);
    _markEconomyAggregatesDirty();
  }

  // Restore divisions
  _provDivIndexDirty=true;_provDivIndex.clear();
  for(const civ of civs){_clearCivRenderMovers(civ);civ.foxes=[];civ._damagedDivs?.clear();}
  for(const dd of (state.divData||[])){
    if(dd.civId < 0 || dd.civId >= civs.length) continue;
    const prov = provinces[dd.provId]; if(!prov) continue;
    const fox  = new Division(dd.civId, dd.provId);
    fox.id=dd.id; fox.hp=dd.hp; fox.maxHp=dd.maxHp;
    fox.size=dd.size; fox.path=dd.path||[];
    fox.autoMove=(dd.civId===state.playerCivId&&playerAutoMoveUniversal)?true:!!dd.autoMove;

    fox._planTargetPid=-1; fox._autoRepathAt=0;
    civs[dd.civId].foxes.push(fox);_syncDamagedDiv(fox);
  }
  let _maxDivId=0;
  for(const civ of civs) for(const fox of civ.foxes) if(fox.id>=_maxDivId) _maxDivId=fox.id+1;
  _divId=_maxDivId;

  // Restore alliances
  alliances=[]; _nextAllianceId=0; _touchAllianceMembership();
  for(const ad of (state.allianceData||[])){
    alliances.push({id:ad.id, name:ad.name||'Alliance', leaderId:ad.leaderId??-1,
      memberIds:new Set(Array.isArray(ad.memberIds)?ad.memberIds:[]),
      color:ad.color||allianceColor(alliances.length)});
  }
  _nextAllianceId = state.nextAllianceId ?? (alliances.length>0 ? Math.max(...alliances.map(a=>a.id))+1 : 0);
  _allianceLabelDirty=true; _allianceBorderDirty=true; _allianceLabelImgCache.clear();

  // Restore date (keep original save's date so world resumes where it left off)
  gameDay=state.gameDay||1; gameMonth=state.gameMonth||1;
  gameYear=state.gameYear||2024; gameDayFrac=state.gameDayFrac||0;

  // Restore bot techs fallback
  initBotPhases();
  for(const civ of civs){
    // Only synthesize defaults for genuinely old/missing save data. An explicit
    // empty Set means this country intentionally had zero researched techs.
    if(!_civsWithSavedTechState.has(civ.id) && (!civ.techs || !civ.techs.size)){
      const lv=MODERN_NATION_TECHS[civ.name];
      civ.techs=new Set(_buildTechSet(lv?lv[0]:3, lv?lv[1]:3));
    }
    _recomputeBotTechBonus(civ);
    civ._techCooldown=civ.id%60;
  }

  // NOTE: playerCivId and gameReady intentionally NOT set here —
  // caller (country bar) will call _startGamePreserveMap with the chosen civId.
}

function buildSaveState(){
  // Serialize province ownership (the core dynamic state)
  const ownerArr = Array.from(provOwner);

  // Serialize civ resources and metadata (including bot techs)
  const civData = civs.map(c=>({
    id: c.id,
    name: c.name,
    color: c.color,
    res: {...c.res},
    isPlayer: c.isPlayer,
    countryId: c.countryId,
    tnoCanonicalName: c.tnoCanonicalName||'',
    customNation: !!c.customNation,
    // Save each country's technology state. For the active player, playerTechs is
    // the authoritative live tree; AI/new states use civ.techs.
    techs: c.id === playerCivId ? [...playerTechs] : (c.techs ? [...c.techs] : []),
    // Legacy field kept so older loaders remain compatible with these saves.
    botTechs: c.id !== playerCivId && c.techs ? [...c.techs] : []
  }));

  // Serialize divisions
  const divData = [];
  for(const civ of civs){
    for(const fox of civ.foxes){
      divData.push({
        id: fox.id,
        civId: fox.civId,
        provId: fox.provId,
        hp: fox.hp,
        maxHp: fox.maxHp,
        size: fox.size,
        path: fox.path ? [...fox.path] : [],
        autoMove: !!fox.autoMove,
        borderMode: fox.borderMode || '',
        borderTargets: [...(fox.borderTargets||[])]
      });
    }
  }

  // Serialize wars
  const wars = Array.from(warPairs);

  // Serialize province building levels
  const provBld = provinces.map(p => ({...p.bld}));

  // Serialize alliances — convert each Set<number> to an Array for JSON
  const allianceData = alliances.map(a => ({
    id:       a.id,
    name:     a.name,
    leaderId: a.leaderId,
    memberIds: [...a.memberIds],
    color:    a.color
  }));

  // Serialize researched player technology. Keep the legacy flat list and also
  // store the two trees explicitly so economic and military progress are durable

  const warRecordData = [...warRecords.values()].map(r=>({a:r.a,b:r.b,startA:[...(r.startA||[])],startB:[...(r.startB||[])],startOwners:{...(r.startOwners||{})},contrib:{...(r.contrib||{})},started:r.started||null,type:r.type||'normal',declaredBy:r.declaredBy,attackerSide:[...(r.attackerSide||[])],defenderSide:[...(r.defenderSide||[])],declarationId:r.declarationId??null,declarationOwners:{...(r.declarationOwners||{})}}));
  const warDeclarationData=[...warDeclarations.values()].map(d=>({id:d.id,type:d.type||'normal',declaredBy:d.declaredBy,attackerSide:[...(d.attackerSide||[])],defenderSide:[...(d.defenderSide||[])],originalOwners:{...(d.originalOwners||{})},invaded:[...(d.invaded||[])],started:d.started||null}));
  const pendingPeaceData = _serializePendingPeaceSettlements();
  const cityOriginalOwners = _cityOriginalOwnerIds ? Array.from(_cityOriginalOwnerIds) : [];
  const destroyedCitySourceIds=[]; // legacy field retained for save compatibility; cities are never destroyable
  const originalMapOwners = _originalMapOwnerIds ? Array.from(_originalMapOwnerIds) : [];

  return {
    version: 1,
    scenario: activeScenario,
    gameMode,
    playerCivId,

    gameSpeed: _preEscSpeed,
    civData,

  };

function loadGameFromText(text){
  // Strip comment lines
  const jsonStr = text.split('\n').filter(l=>!l.startsWith('//')).join('\n');
  const state = JSON.parse(jsonStr);
  if(!state || state.version !== 1) throw new Error('Invalid or unsupported save format');

  // Ensure map is ready
  if(!mapReady) throw new Error('Map not loaded yet — wait for loading to finish');

  // Wipe all state from any previous game first
  _fullReset();

  // Restore scenario but DON'T apply — ownership is fully restored from ownerArr
  activeScenario = state.scenario || 'modern';
  gameMode = state.gameMode || 'spectator';
  playerAutoMoveUniversal=!!state.playerAutoMoveUniversal;
  if(state.botDifficulty&&BOT_DIFFICULTY_PROFILES[state.botDifficulty]){botDifficulty=state.botDifficulty;const bd=document.getElementById('set-bot-difficulty');if(bd)bd.value=botDifficulty;}

  // Recreate released/custom nations before ownerArr references their ids.
  for(const cd of [...(state.civData||[])].sort((a,b)=>(a.id??0)-(b.id??0))){
    if((cd.id??-1)>=civs.length) _restoreDynamicCivFromSave(cd);
  }
  // Restore names/colors for both original and custom civs. Old subject metadata is intentionally ignored.
  for(const cd of (state.civData||[])) if(cd.id>=0&&cd.id<civs.length) _restoreDynamicCivFromSave(cd);

  // Restore ownership
  const saved = state.ownerArr;
  if(saved && saved.length === provOwner.length){
    // Clear all
    for(let pid=0; pid<provinces.length; pid++){
      const cur = provOwner[pid];
      if(cur >= 0) { restoreProvUnownedColor(pid); }
    }
    for(const civ of civs){ civ.ownedProvs.clear(); civ.territory.resetProvinces?.([]); }
    provOwner.fill(-1);

    for(let pid=0; pid<saved.length; pid++){
      const civId = saved[pid];
      if(civId >= 0 && civId < civs.length){
        provOwner[pid] = civId;
        civs[civId].ownedProvs.add(pid);
        bakeProvOwnedColor(pid, civId);
      }
    }
  }

  _restoreOriginalMapOwners(state.originalMapOwners);
  _restoreCityOriginalOwners(state.cityOriginalOwners);
  _destroyedCitySourceIds.clear();_markCapitalsDirty(); // migrate legacy saves: all city markers are restored permanently

  // Restore resources and per-country technology sets. The player tree is
  // promoted into playerTechs after playerCivId is restored below.
  const _savedPlayerTechs = _deserializeSavedPlayerTechs(state);
  const _civsWithSavedTechState = new Set();
  for(const cd of (state.civData||[])){
    if(cd.id >= 0 && cd.id < civs.length){
      Object.assign(civs[cd.id].res, cd.res);
      const hasPlayerTree = cd.id === state.playerCivId &&
        ((state.techTreeData && (Array.isArray(state.techTreeData.econ) || Array.isArray(state.techTreeData.mil))) || Array.isArray(state.techData));
      const hasExplicitTechs = Array.isArray(cd.techs) || hasPlayerTree || Array.isArray(cd.botTechs);
      const savedTechs = Array.isArray(cd.techs) ? cd.techs
        : (hasPlayerTree ? _savedPlayerTechs
        : (Array.isArray(cd.botTechs) ? cd.botTechs : []));
      civs[cd.id].techs = new Set(_filterKnownTechIds(savedTechs));
      if(hasExplicitTechs) _civsWithSavedTechState.add(cd.id);
      _recomputeBotTechBonus(civs[cd.id]);
    }
  }

  // Restore wars
  warPairs.clear();_markWarAdjDirty();
  for(const w of (state.wars||[])){
    // Migrate old string-format keys ("3-17") to integer format (3*10000+17)
    if(typeof w === 'string'){
      const parts = w.split('-');
      if(parts.length===2){
        const a=parseInt(parts[0]), b=parseInt(parts[1]);
        if(!isNaN(a)&&!isNaN(b)) warPairs.add(a<b ? a*10000+b : b*10000+a);_markWarAdjDirty();
      }
    } else {
      warPairs.add(w);_markWarAdjDirty(); // already integer
    }
  }
  warRecords.clear();
  for(const wr of (state.warRecordData||[])){
    if(wr&&wr.a>=0&&wr.b>=0) warRecords.set(warKey(wr.a,wr.b),{key:warKey(wr.a,wr.b),a:wr.a,b:wr.b,startA:[...(wr.startA||[])],startB:[...(wr.startB||[])],startOwners:{...(wr.startOwners||{})},contrib:{...(wr.contrib||{})},started:wr.started||null,type:WAR_TYPES[wr.type]?wr.type:'normal',declaredBy:Number.isInteger(wr.declaredBy)?wr.declaredBy:wr.a,attackerSide:[...(wr.attackerSide||[wr.a])],defenderSide:[...(wr.defenderSide||[wr.b])],declarationId:Number.isInteger(wr.declarationId)?wr.declarationId:null,declarationOwners:{...(wr.declarationOwners||{})}});
  }
  // Old saves still get functional conference snapshots from their current war state.
  for(const key of warPairs){const [a,b]=_decodeWarKey(key);if(!warRecords.has(key))_ensureWarRecord(a,b);}
  warDeclarations.clear();let maxDecl=0;
  for(const wd of (state.warDeclarationData||[])){if(!wd||!Number.isInteger(wd.id))continue;warDeclarations.set(wd.id,{id:wd.id,type:WAR_TYPES[wd.type]?wd.type:'normal',declaredBy:wd.declaredBy,attackerSide:[...(wd.attackerSide||[])],defenderSide:[...(wd.defenderSide||[])],originalOwners:{...(wd.originalOwners||{})},invaded:new Set((wd.invaded||[]).filter(Number.isInteger)),started:wd.started||null});maxDecl=Math.max(maxDecl,wd.id);}
  // Migration for saves made before declaration objects existed.
  for(const rec of warRecords.values())if(rec.declarationId!=null&&!warDeclarations.has(rec.declarationId)){const inv=new Set();for(const [pidStr,owner] of Object.entries(rec.declarationOwners||{})){const pid=+pidStr;if(provinces[pid]&&provOwner[pid]!==+owner)inv.add(pid);}warDeclarations.set(rec.declarationId,{id:rec.declarationId,type:rec.type,declaredBy:rec.declaredBy,attackerSide:[...(rec.attackerSide||[])],defenderSide:[...(rec.defenderSide||[])],originalOwners:{...(rec.declarationOwners||{})},invaded:inv,started:rec.started||null});maxDecl=Math.max(maxDecl,rec.declarationId);}
  _warDeclarationSerial=Math.max(1,maxDecl+1);
  _restorePendingPeaceSettlements(state.pendingPeaceData||[]);

  // Restore province buildings
  if(state.provBld && state.provBld.length === provinces.length){
    for(let pid=0; pid<provinces.length; pid++){
      Object.assign(provinces[pid].bld, state.provBld[pid]);
    }
    _markEconomyAggregatesDirty();
  }

  // Restore player
  playerCivId = state.playerCivId ?? -1;
  if(playerCivId >= 0 && playerCivId < civs.length){
    civs[playerCivId].isPlayer = true;
    const capPid=_capitalProvinceForCiv(playerCivId);
    if(capPid>=0){
      const cap = provinces[capPid];
      if(cap){ camX=cap.centerTx-canvas.width/(2*TS*zoom); camY=cap.centerTy-canvas.height/(2*TS*zoom); clampCam(); }
    }
    document.getElementById('hud-dot').style.background = civs[playerCivId].color;
    document.getElementById('hud-civname').textContent = civs[playerCivId].name.toUpperCase();
  } else {
    document.getElementById('hud-dot').style.background = 'transparent';
    document.getElementById('hud-civname').textContent = 'WORLD SANDBOX';
  }
  document.getElementById('hud-mode').textContent = SCENARIOS[activeScenario]?.label || '';

  // Restore date
  gameDay = state.gameDay || 1;
  gameMonth = state.gameMonth || 1;
  gameYear = state.gameYear || 2024;
  gameDayFrac = state.gameDayFrac || 0;

  // Restore divisions (foxes)
  _provDivIndexDirty=true;_provDivIndex.clear();
  for(const civ of civs){_clearCivRenderMovers(civ);civ.foxes=[];civ._damagedDivs?.clear();}
  for(const dd of (state.divData||[])){
    if(dd.civId < 0 || dd.civId >= civs.length) continue;
    const civ = civs[dd.civId];
    const prov = provinces[dd.provId];
    if(!prov) continue;
    const fox = new Division(dd.civId, dd.provId);
    fox.id = dd.id;
    fox.hp = dd.hp;
    fox.maxHp = dd.maxHp;
    fox.size = dd.size;
    fox.path = dd.path || [];
    fox.autoMove = (dd.civId===state.playerCivId&&playerAutoMoveUniversal)?true:!!dd.autoMove;
    fox.borderMode = dd.borderMode || '';
    fox.borderTargets = Array.isArray(dd.borderTargets) ? dd.borderTargets.filter(Number.isInteger) : [];
    fox._planTargetPid = -1;
    fox._autoRepathAt = 0;
    civ.foxes.push(fox);_syncDamagedDiv(fox);
  }

  // Sync _divId so newly created divisions never collide with restored ones
  let _maxDivId = 0;
  for(const civ of civs) for(const fox of civ.foxes) if(fox.id >= _maxDivId) _maxDivId = fox.id + 1;
  _divId = _maxDivId;
  for(const civ of civs)if(civ&&civ.id!==state.playerCivId)_botNormalizeWarStacks(civ);

  // Restore alliances — rebuild Sets from saved Arrays
  alliances = []; _touchAllianceMembership();
  _nextAllianceId = 0;
  for(const ad of (state.allianceData || [])){
    alliances.push({
      id:       ad.id,
      name:     ad.name     || 'Alliance',
      leaderId: ad.leaderId ?? -1,
      memberIds: new Set(Array.isArray(ad.memberIds) ? ad.memberIds : []),
      color:    ad.color    || allianceColor(alliances.length)
    });
  }
  // Restore the ID counter so new alliances get fresh IDs
  _nextAllianceId = state.nextAllianceId ?? (alliances.length > 0
    ? Math.max(...alliances.map(a => a.id)) + 1
    : 0);
  // Invalidate all alliance render caches so map reflects loaded state
  _allianceLabelDirty  = true;
  _allianceBorderDirty = true;
  _allianceLabelImgCache.clear();

  // Restore the player's exact economic + military tree and recompute every
  // derived modifier from those unlocked nodes. Old saves using only techData
  // remain supported.
  playerTechs.clear();
  for(const id of _deserializeSavedPlayerTechs(state)) playerTechs.add(id);
  _recomputeTechBonuses();
  _syncPlayerTechsToCiv();
  _divCapCache = -1;
  _clearFoxSpriteCache();

  initBotPhases();
  // Ensure every bot has tech structures (covers saves made before bot-tech system)
  for(const civ of civs){
    if(civ.id === playerCivId) continue;
    if(!_civsWithSavedTechState.has(civ.id) && (!civ.techs || !civ.techs.size)){
      const level = MODERN_NATION_TECHS[civ.name];
      const econT = level ? level[0] : 3;
      const milT  = level ? level[1] : 3;
      civ.techs = new Set(_buildTechSet(econT, milT));
    }
    _recomputeBotTechBonus(civ);
    // Prime cooldown so research resumes promptly after load
    civ._techCooldown = civ.id % 60;
  }
  gameReady = true;
  document.body.classList.add('game-running');
  AudioEngine.init(); AudioEngine.syncFromSettings(); AudioEngine.load();
  setGameSpeed(0); // start paused
  document.getElementById('top-bar').classList.add('game-visible');
  document.getElementById('gen-bar').classList.add('game-visible');
  setTimeout(()=>{ buildGenBar(); updateFlagBanner(); buildAllianceWidget(); updateMilBar(); _tryStartQueuedPeaceConferences(); }, 50);
}

  if(keysDown.size){clampCam();_markCameraMotion();}
}

          buildCountryBar(document.getElementById('cb-search')?.value || '');
          // Pan camera to civ capital
          const capPid=_capitalProvinceForCiv(civId);
          if(capPid>=0){
            const cap = provinces[capPid];
            if(cap){
              camX = cap.centerTx - canvas.width/(2*TS*zoom);
              camY = cap.centerTy - canvas.height/(2*TS*zoom);
              clampCam();
            }
          }
          // Scroll the active item into view in the sidebar
          setTimeout(()=>{
            const activeItem = document.querySelector('#cb-list .cb-item.active');
            if(activeItem) activeItem.scrollIntoView({block:'nearest',behavior:'smooth'});
          }, 50);
        }
      }
    }
    return;
  }

  const hadSel  = selectedDivs.size > 0;
  const selDivs = playerDivs().filter(d=>d.selected);

  // Only block click-on-division when NOT issuing a move order.
  // If the player has units selected, the click is a move destination, not a unit-select.
  if(selDivs.length === 0){
    for(const d of playerDivs()) if(d.hitTest(mx,my)) return;
  }

  const {tx,ty} = screenToTile(mx, my);
  const tile    = tileAt(tx,ty);

  // Check if click lands on a sea province tile (water tile with a province assigned)
  const clickPid = (tx>=0&&ty>=0&&tx<WORLD_W&&ty<WORLD_H) ? tileProvMap[ty*WORLD_W+tx] : -1;
  const clickIsSea = clickPid>=0 && provinces[clickPid] && provinces[clickPid].isSea;

  if(!tile || (tile.type==='water' && !clickIsSea)){
    selectedDivs.clear(); updateMilBar(); closePanel(); return;
  }
  const pid = clickIsSea ? clickPid : (tile.type==='water' ? -1 : tileProvMap[ty*WORLD_W+tx]);
  if(pid < 0){ selectedDivs.clear(); updateMilBar(); closePanel(); return; }

  if(selDivs.length > 0){
    const destOwner = provOwner[pid];
    const destIsSea = provinces[pid]&&provinces[pid].isSea;
    const blocked = !destIsSea && destOwner>=0 && destOwner!==playerCivId && !areAtWar(playerCivId,destOwner) && !areAllied(playerCivId,destOwner);
    if(blocked){
      // Destination is foreign neutral territory — just open its panel
      AudioEngine.error();openPanel(pid);
      return;
    }
    // Visual grouping is draw-only. Every selected underlying division remains a
    // real independent unit, so a move order is applied to every selected unit.
    for(const d of selDivs){
      if(d.provId===pid)continue;
      _disableAutoMoveFor(d);
      d._vGroupId=null;
      const path=bfsProvPathRestricted(d.provId,pid,playerCivId);
      if(path&&path.length>1)d.path=path.slice(1);
    }
    const mp=provinces[pid];if(mp)AudioEngine.moveOrder(mp.centerTx,mp.centerTy,selDivs.length);
    selectedDivs.clear(); updateMilBar(); return;
  }

  if(hadSel){ updateMilBar(); return; }
  if(pid === selectedProv){
    // Clicking same province again: if it's a civ territory, keep panel open
    // If it's neutral and panel is open, close it
    const o=provOwner[pid];
    if(o<0&&gameMode!=='sandbox'){ closePanel(); return; }
    if(panelCivId>=0&&o>=0&&o===panelCivId){ closePanel(); return; }
  }
  // Don't open panel for neutral territory unless player can claim it or sandbox
  // Sea zones never get a panel (they can't be claimed or owned)
  if(provinces[pid]&&provinces[pid].isSea){ selectedDivs.clear(); updateMilBar(); closePanel(); return; }  const pidOwner=provOwner[pid];
  if(pidOwner<0&&gameMode!=='sandbox'){
    const canClaim=isAdjToOwned(pid)||ownedProvs().size===0;
    if(!canClaim){ selectedDivs.clear(); updateMilBar(); closePanel(); return; }
  }
  AudioEngine.select();openPanel(pid);
}

function _screenPid(clientX, clientY){

  // Return pid for any assigned province, including sea provinces
  return pid;
}

      }

      _boxSelecting = false;
    }
  }

// Mass upgrade a single building type across all owned provinces (best terrain first)
function massUpgradeBuilding(bname){
  if(playerCivId<0) return;
  const r=civs[playerCivId].res,def=BLDEFS[bname],owned=civs[playerCivId].ownedProvs;
  let upgraded=0,changed=true;
  const byLv=[[],[],[]];
  // Same semantics as the previous stable score sort: snapshot affordable candidates for
  // this pass, then process level 0 before 1 before 2 while preserving ownedProvs order.
  while(changed){
    changed=false;byLv[0].length=byLv[1].length=byLv[2].length=0;
    for(const pid of owned){
      const p=provinces[pid],lv=p.bld[bname];if(lv>=3)continue;
      const cost=def.costs[lv];if(r.wood<cost.w||r.stone<cost.s)continue;byLv[lv].push(pid);
    }
    for(let lv=0;lv<3;lv++){
      const cost=def.costs[lv],a=byLv[lv];
      for(let i=0;i<a.length;i++){
        const pid=a[i];if(r.wood<cost.w||r.stone<cost.s)continue;
        r.wood-=cost.w;r.stone-=cost.s;provinces[pid].bld[bname]++;_markEconomyAggregatesDirty();
        burst(tcx(provinces[pid].centerTx),tcy(provinces[pid].centerTy),def.tab==='econ'?'#6ad060':'#e05050');
        upgraded++;changed=true;
      }
    }
  }
  if(upgraded>0){
    _markUpgradeMetaDirty();
    AudioEngine.industry();
    if(selectedProv>=0) openPanel(selectedProv);
    buildGenBar();
  }
}

// Mass upgrade ALL building types until resources run out
const _massAllBlds=[...ECON_BLDS,...MIL_BLDS],_massAllBuckets=[[],[],[],[],[],[]];
// Preserve the old stable score sort exactly, without allocating candidate objects or sorting.
// Bucket order is: econ L0, mil L0, econ L1, mil L1, econ L2, mil L2.
function massUpgradeAll(){
  if(playerCivId<0) return;
  const owned=civs[playerCivId].ownedProvs,r=civs[playerCivId].res,buckets=_massAllBuckets;
  let anyChanged=true,totalChanged=false;
  while(anyChanged){
    anyChanged=false;
    for(let i=0;i<6;i++)buckets[i].length=0;
    for(let bi=0;bi<_massAllBlds.length;bi++){
      const bname=_massAllBlds[bi],def=BLDEFS[bname],isEcon=def.tab==='econ';
      for(const pid of owned){
        const p=provinces[pid],lv=p.bld[bname];if(lv>=3)continue;
        const cost=def.costs[lv];if(r.wood<cost.w||r.stone<cost.s)continue;
        // Scores from the old formula are strictly ordered by this bucket mapping.
        const bucket=lv*2+(isEcon?0:1);buckets[bucket].push(bi,pid);
      }
    }
    let count=0;for(let i=0;i<6;i++)count+=buckets[i].length;if(!count)break;
    for(let bucket=0;bucket<6;bucket++){
      const a=buckets[bucket],lv=bucket>>1;
      for(let i=0;i<a.length;i+=2){
        const bi=a[i],pid=a[i+1],bname=_massAllBlds[bi],def=BLDEFS[bname],cost=def.costs[lv];
        if(r.wood<cost.w||r.stone<cost.s)continue;
        r.wood-=cost.w;r.stone-=cost.s;provinces[pid].bld[bname]++;_markEconomyAggregatesDirty();
        burst(tcx(provinces[pid].centerTx),tcy(provinces[pid].centerTy),def.tab==='econ'?'#6ad060':'#e05050');
        anyChanged=true;totalChanged=true;
      }
    }
  }
  if(totalChanged)_markUpgradeMetaDirty();
  if(selectedProv>=0) openPanel(selectedProv);
  buildGenBar();
}

// Player's researched tech IDs and bonus cache are declared at the top of the script.
// _recomputeTechBonuses() resets and recomputes them from scratch.

function _knownTechIdsByTree(){
  const out = {econ:new Set(), mil:new Set()};
  for(const key of ['econ','mil']){
    const tree = TECH_TREES[key];
    if(!tree) continue;
    for(const row of tree.rows) for(const t of row.techs) out[key].add(t.id);
  }
  return out;
}

function _filterKnownTechIds(ids){
  const known = _knownTechIdsByTree();
  const all = new Set([...known.econ, ...known.mil]);
  const out = [];
  const seen = new Set();
  for(const id of (Array.isArray(ids) ? ids : [])){
    if(typeof id !== 'string' || !all.has(id) || seen.has(id)) continue;
    seen.add(id); out.push(id);
  }
  return out;
}

function _serializePlayerTechTrees(){
  const known = _knownTechIdsByTree();
  return {
    econ:[...playerTechs].filter(id=>known.econ.has(id)),
    mil:[...playerTechs].filter(id=>known.mil.has(id))
  };
}

function _deserializeSavedPlayerTechs(state){
  const tree = state && state.techTreeData;
  if(tree && (Array.isArray(tree.econ) || Array.isArray(tree.mil))){
    return _filterKnownTechIds([...(tree.econ||[]), ...(tree.mil||[])]);
  }
  return _filterKnownTechIds(state && Array.isArray(state.techData) ? state.techData : []);
}

function _syncPlayerTechsToCiv(){
  if(playerCivId < 0 || playerCivId >= civs.length) return;
  civs[playerCivId].techs = new Set(_filterKnownTechIds([...playerTechs]));
  // Keep the AI-style cache valid too, so choosing a different country after
  // loading/spectating never loses this nation's saved technology modifiers.
  _recomputeBotTechBonus(civs[playerCivId]);
}

function _recomputeTechBonuses(){
  // Reset
  for(const k in techBonus) techBonus[k] = 0;
  // Accumulate all researched techs
  for(const tree of Object.values(TECH_TREES)){
    for(const row of tree.rows){
      for(const t of row.techs){
        if(!playerTechs.has(t.id)) continue;
        for(const [k, v] of Object.entries(t.bonus)){
          techBonus[k] = (techBonus[k] || 0) + v;
        }
      }
    }
  }  if(playerCivId>=0&&civs[playerCivId]){civs[playerCivId]._moveMul=1+(techBonus.movMul||0);civs[playerCivId]._econRateDirty=true;}
}

// Returns true if the tier's requirements are met (all required tech IDs researched)
function _tierUnlocked(row){
  if(!row.requires) return true;
  return row.requires.every(id => playerTechs.has(id));
}

      if(!bonus) return '';

      const mns = v => '−'+Math.round(v*100)+'%';
      if(bonus.atkMul)    pills.push(`<span class="tech-bp atk">${pct(bonus.atkMul)} ATK</span>`);
      if(bonus.defMul)    pills.push(`<span class="tech-bp def">${pct(bonus.defMul)} DEF</span>`);
      if(bonus.dmgReduce) pills.push(`<span class="tech-bp dmg">${mns(bonus.dmgReduce)} DMG</span>`);
      if(bonus.movMul)    pills.push(`<span class="tech-bp mov">${pct(bonus.movMul)} MOV</span>`);
      if(bonus.capBonus)  pills.push(`<span class="tech-bp cap">+${bonus.capBonus} CAP</span>`);
      if(bonus.allMul)    pills.push(`<span class="tech-bp eco">${pct(bonus.allMul)} ALL</span>`);
      if(bonus.foodMul)   pills.push(`<span class="tech-bp eco">${pct(bonus.foodMul)} 🌾</span>`);
      if(bonus.woodMul)   pills.push(`<span class="tech-bp eco">${pct(bonus.woodMul)} 🪵</span>`);
      if(bonus.stoneMul)  pills.push(`<span class="tech-bp eco">${pct(bonus.stoneMul)} 🪨</span>`);
      if(bonus.coinsMul)  pills.push(`<span class="tech-bp eco">${pct(bonus.coinsMul)} 🪙</span>`);
      if(bonus.fortBonus) pills.push(`<span class="tech-bp def">+${bonus.fortBonus} FORT</span>`);
      return pills.join('');
    }

    // Helper: render a single tech node — compact fixed-height card
    function nodeHTML(tech, unlocked, nodeClass, tier=0){
      const researched = playerTechs.has(tech.id);
      const canAffordTech = r && r.coins >= tech.cost;
      const cls = [
        researched ? 'researched' : (!unlocked ? 'locked' : 'available'),
        (!researched && unlocked && canAffordTech) ? 'afford' : ''
      ].join(' ').trim();
      const tooltip = tech.desc + (researched ? '' : (' | Cost: '+tech.cost+' coins'+(!unlocked?' | 🔒 Locked':'')));
      const costHtml = researched
        ? `<span class="tech-cost researched-lbl">✓ DONE</span>`
        : `<span class="tech-cost ${canAffordTech ? 'can-afford' : 'cant-afford'}">🪙 ${tech.cost.toLocaleString()}</span>`;
      const lockIcon = (!researched && !unlocked) ? `<span style="position:absolute;top:6px;right:7px;font-size:10px;opacity:.5">🔒</span>` : '';
      return `<div class="${nodeClass} ${cls} ${tier>=10?'tech-tier-'+tier:''}" data-techid="${tech.id}" data-treeid="${treeKey}" title="${tooltip}">
        ${lockIcon}
        <div class="tech-card-top">
          <span class="tech-icon">${tech.icon}</span>
          <div class="tech-card-title">
            <div class="tech-name">${tech.name}</div>
          </div>
        </div>
        <div class="tech-bonus-row">${bonusPills(tech.bonus)}</div>
        <div class="tech-card-foot">${costHtml}</div>
      </div>`;
    }

    let html = '';

    if(treeKey === 'mil'){
      // ── Military: 6-column grid (Armor | Guns | Special Tools | Boots | Air Power | WMD) ──
      const COLS = [
        { key:'armor', label:'⚔ ARMOR',         ids:['arm1','arm2','arm3','arm4','arm5','arm6','arm7','arm8','arm9','arm10','arm11','arm12'] },
        { key:'guns',  label:'🔫 WEAPONS',       ids:['gun1','gun2','gun3','gun4','gun5','gun6','gun7','gun8','gun9','gun10','gun11','gun12'] },
        { key:'spc',   label:'🪤 SPECIAL TOOLS', ids:['spc1','spc2','spc3','spc4','spc5','spc6','spc7','spc8','spc9','spc10','spc11','spc12'] },
        { key:'boots', label:'🥾 MOBILITY',      ids:['bot1','bot2','bot3','bot4','bot5','bot6','bot7','bot8','bot9','bot10','bot11','bot12'] },
        { key:'air',   label:'✈️ AIR POWER',     ids:['air1','air2','air3','air4','air5','air6','air7','air8','air9','air10','air11','air12'] },
        { key:'wmd',   label:'☢️ WMD',           ids:['wmd1','wmd2','wmd3','wmd4','wmd5','wmd6','wmd7','wmd8','wmd9','wmd10','wmd11','wmd12'] },
      ];

      // Build id→row lookup for tier unlock checks
      const idToRow = new Map();
      for(const row of tree.rows) for(const t of row.techs) idToRow.set(t.id, row);
      // Build id→tech lookup
      const idToTech = new Map();
      for(const row of tree.rows) for(const t of row.techs) idToTech.set(t.id, t);

      html += `<div class="tech-mil-grid">`;
      for(const col of COLS){
        html += `<div class="tech-mil-col">`;
        html += `<div class="tech-mil-col-label">${col.label}</div>`;
        for(let i = 0; i < col.ids.length; i++){
          const techId = col.ids[i];
          const tech = idToTech.get(techId);
          if(!tech) continue;
          const row = idToRow.get(techId);
          const unlocked = _tierUnlocked(row);
          if(i > 0) html += `<div class="tech-mil-arrow">▼</div>`;
          html += nodeHTML(tech, unlocked, 'tech-mil-node', i+1);
        }
        html += `</div>`;
      }
      html += `</div>`;

    } else {
      // ── Economic: stacked rows (original layout) ──
      for(let ri=0;ri<tree.rows.length;ri++){
        const row=tree.rows[ri], unlocked = _tierUnlocked(row), tier=ri+1;
        html += `<div class="tech-row-label ${tier>=10?'tier-'+tier:''}">${row.label}</div><div class="tech-row">`;
        for(const tech of row.techs){
          html += nodeHTML(tech, unlocked, 'tech-node', tier);
        }
        html += `</div>`;
      }
    }

    el.innerHTML = html;

    // Wire click events — covers both .tech-node and .tech-mil-node
    el.querySelectorAll('.tech-node.available:not(.researched), .tech-mil-node.available:not(.researched)').forEach(node => {
      node.addEventListener('click', () => {
        const techId = node.dataset.techid;
        const tId = node.dataset.treeid;
        let techDef = null;
        outer: for(const row2 of TECH_TREES[tId].rows){
          for(const t of row2.techs){ if(t.id === techId){ techDef = t; break outer; } }
        }
        if(!techDef || playerTechs.has(techId)) return;
        const res = playerCivId >= 0 ? civs[playerCivId].res : null;
        if(!res || res.coins < techDef.cost) return;
        res.coins -= techDef.cost;
        playerTechs.add(techId);
        AudioEngine.tech(tId);
        _recomputeTechBonuses();
        _syncPlayerTechsToCiv();
        _clearFoxSpriteCache(); // tier visuals may have changed
        _divCapCache = -1;
        buildTechTreePanels();
        updateTechCoinDisplay();
      });
    });
  }
}

function advanceDate(ticks){
  gameDayFrac+=ticks*0.04; // ~25 real-seconds per in-game month at speed 1
  let changed=false;
  while(gameDayFrac>=1){
    gameDayFrac-=1;gameDay++;changed=true;
    const daysInMonth=(gameMonth===2&&((gameYear%4===0&&gameYear%100!==0)||gameYear%400===0))?29:(_DAYS_IN_MONTH[gameMonth-1]||30);
    if(gameDay>daysInMonth){gameDay=1;gameMonth++;}

  if((_dateHudDirty||changed)&&_timeDateEl){

    _timeDateEl.textContent=`${gameDay} ${MONTHS[gameMonth-1]} ${yearStr}`;_dateHudDirty=false;
  }
}

// FOLDER PERSISTENCE  — IndexedDB stores directory handles so the
// game reconnects to Saves + Start folders silently on every load.
// First visit: pick folder once via the UI button (required by the
// browser security model).  Every subsequent visit: handles are
// restored from IndexedDB and permissions re-verified automatically
// — no picker dialog, no confirmation needed.
// ══════════════════════════════════════════════════════════════════
const _IDB_NAME  = 'WorldSandboxFS';
const _IDB_VER   = 1;
const _IDB_STORE = 'handles';

function _openIDB(){
  return new Promise((res,rej)=>{
    const req = indexedDB.open(_IDB_NAME, _IDB_VER);
    req.onupgradeneeded = e => e.target.result.createObjectStore(_IDB_STORE);
    req.onsuccess  = e => res(e.target.result);
    req.onerror    = e => rej(e.target.error);
  });
}
async function _idbSet(key, value){
  try{
    const db = await _openIDB();
    return new Promise((res,rej)=>{
      const tx = db.transaction(_IDB_STORE,'readwrite');
      tx.objectStore(_IDB_STORE).put(value, key);
      tx.oncomplete = ()=>res();
      tx.onerror    = e=>rej(e.target.error);
    });
  }catch(e){ console.warn('IDB set failed:',e); }
}
async function _idbGet(key){
  try{
    const db = await _openIDB();
    return new Promise((res,rej)=>{
      const tx  = db.transaction(_IDB_STORE,'readonly');
      const req = tx.objectStore(_IDB_STORE).get(key);
      req.onsuccess = e => res(e.target.result ?? null);
      req.onerror   = e => rej(e.target.error);
    });
  }catch(e){ return null; }
}

// Re-verify a stored handle's permission without a picker dialog.
// 'granted'  → already live, no UI needed.
// 'prompt'   → browser needs a micro-prompt (tiny banner, no picker).
// Returns true if we end up with a live grant.
async function _ensurePermission(handle, mode='readwrite'){
  if(!handle) return false;
  try{
    let perm = await handle.queryPermission({mode});
    if(perm === 'granted') return true;
    perm = await handle.requestPermission({mode});
    return perm === 'granted';
  }catch(e){ return false; }
}

// ── Startup: silently restore stored handles ──────────────────────
async function _restoreHandles(){
  const [startH, savesH] = await Promise.all([
    _idbGet('startDir'),
    _idbGet('savesDir')
  ]);

  if(startH){
    const ok = await _ensurePermission(startH, 'read');
    if(ok){
      _startDirHandle = startH;
      try{
        await loadStartFolderCards(startH);
      }catch(e){}
    }
  }

  if(savesH){
    const ok = await _ensurePermission(savesH, 'readwrite');
    if(ok){
      _savesDirHandle      = savesH;
      _savesDirWriteHandle = savesH;
      document.getElementById('sp-load-bar-desc').textContent =
        `\u2713 Auto-linked: "${savesH.name}" \u2014 click to load a save`;
      document.getElementById('sm-folder-status').textContent =
        `\uD83D\uDCC1 "${savesH.name}" (auto-linked)`;
      document.getElementById('sm-save-folder-btn').disabled = false;
    }
  }
}

// ── Coordinate → province ─────────────────────────────────────────
function _edScreenToPid(clientX, clientY){
  const pt=clientToCanvasPoint(clientX,clientY);
  const {tx, ty}=screenToTile(pt.x,pt.y);
  if(tx < 0 || ty < 0 || tx >= WORLD_W || ty >= WORLD_H) return -1;
  return tileProvMap[ty * WORLD_W + tx];
}

// ── Brush: collect provinces within radius steps of pid ───────────
function _edBrushProvs(pid){
  if(pid < 0) return [];
  if(editorBrush === 1) return [pid];
  // BFS over province adjacency graph (provAdj Map) up to editorBrush-1 hops
  const visited = new Set([pid]);
  let frontier  = [pid];
  for(let i = 1; i < editorBrush; i++){
    const next = [];
    for(const p of frontier){
      const nbrs = provAdj.get(p);
      if(!nbrs) continue;
      for(const n of nbrs){
        if(!visited.has(n) && provinces[n] && !provinces[n].isSea){
          visited.add(n); next.push(n);
        }
      }
    }
    if(!next.length) break;
    frontier = next;
  }
  return [...visited];
}

// ── Paint one province (or brush set) ────────────────────────────
function _edPaintProv(pid){
  if(pid < 0) return;
  if(!provinces[pid] || provinces[pid].isSea) return;

  // For brush>1 we don't skip on _edLastPid (let brush paint all)
  if(editorBrush === 1 && pid === _edLastPid) return;
  _edLastPid = pid;

  const targets = _edBrushProvs(pid);
  let painted = false;

  for(const p of targets){
    if(!provinces[p] || provinces[p].isSea) continue;

    if(editorTool === 'nation'){
      if(editorCivId < 0 || editorCivId >= civs.length){
        _edSetStatus('Select a nation from the left bar first!', false);
        _edUpdateInstruction();
        return;
      }
      const prev = provOwner[p];
      if(prev === editorCivId) continue;
      if(prev >= 0 && civs[prev]){
        civs[prev].ownedProvs.delete(p);
        civs[prev].territory.deleteProvince?.(p);
      }
      provOwner[p] = editorCivId;
      civs[editorCivId].ownedProvs.add(p);
      civs[editorCivId].territory.addProvince?.(p);
      bakeProvOwnedColor(p, editorCivId);
      painted = true;
    } else {
      const prev = provOwner[p];
      if(prev < 0) continue;
      civs[prev].ownedProvs.delete(p);
      civs[prev].territory.deleteProvince?.(p);
      provOwner[p] = -1;
      restoreProvUnownedColor(p);
      painted = true;
    }
  }

  if(painted){
    _markWorldFullDirty(); _terrBorderDirty = true;
    _civLabelDirty = true; _allianceBorderDirty = true;
    _hlDirty = true; _terrBorderAll = null; _terrBorderPlayer = null;
  }
}

// ── Hover highlight drawn each frame by draw() ────────────────────
// We add an editor-specific step at the end of the draw pipeline.
function _edDrawHover(){
  if(!editorMode || _edHoverPid < 0) return;
  if(!provinces[_edHoverPid] || provinces[_edHoverPid].isSea) return;

  const tpx = TS * zoom;
  const sw = canvas.width, sh = canvas.height;

  // All provinces in brush radius
  const targets   = _edBrushProvs(_edHoverPid);
  const targetSet = new Set(targets);

  const isErase  = editorTool === 'erase';
  const glowCol  = isErase ? '#ff4040'
    : (editorCivId >= 0 && civs[editorCivId] ? civs[editorCivId].color : '#ffffff');

  ctx.save();

  // ── 1. Fill tiles of every brush province ─────────────────────
  const fillColor = isErase
    ? 'rgba(255,60,60,0.28)'
    : (editorCivId >= 0 && civs[editorCivId]
        ? glowCol + '55'   // ~33% alpha hex
        : 'rgba(255,255,255,0.18)');

  ctx.fillStyle = fillColor;
  for(const pid of targets){
    const p = provinces[pid];
    if(!p || p.isSea) continue;
    for(const ti of p.tiles){
      const tx2 = ti % WORLD_W, ty2 = (ti / WORLD_W) | 0;
      const sx = (tx2 - camX) * tpx, sy = (ty2 - camY) * tpx;
      if(sx < -tpx || sy < -tpx || sx > sw + tpx || sy > sh + tpx) continue;
      ctx.fillRect(sx, sy, tpx, tpx);
    }
  }

  // ── 2. Build the OUTER perimeter of the whole brush group ──────
  // Collect outline segments from every province in the group,
  // but ONLY keep edges that do NOT border another group member.
  // That gives us the true outer boundary.
  const outerSegs = []; // [x1,y1,x2,y2, ...]

  for(const pid of targets){
    let segs = _provOutlineCache.get(pid);
    if(!segs) segs = _buildProvOutline(pid);

    // Each segment in _buildProvOutline separates this province from
    // an adjacent province. We need to know which adjacent province
    // the segment borders. We use the segment's midpoint tile to
    // find the neighbour.
    // Simpler reliable approach: keep all segments from a province that
    // are NOT shared with another group member.
    // We check by seeing if the tile just across the segment edge
    // (one tile in the perpendicular direction) belongs to a group member.
    for(let i = 0; i < segs.length; i += 4){
      const x1 = segs[i], y1 = segs[i+1];
      const x2 = segs[i+2], y2 = segs[i+3];
      // Midpoint of the segment (in tile coords)
      const mx = (x1 + x2) * 0.5, my = (y1 + y2) * 0.5;
      // Determine the perpendicular offset (into the neighbour's tile)
      // Horizontal edge (y1===y2): neighbour is one tile above or below
      // Vertical edge (x1===x2): neighbour is one tile left or right
      const isH = Math.abs(y2 - y1) < 0.01; // horizontal segment → vertical border
      const dx = isH ? 0 : 0.5, dy = isH ? 0.5 : 0;
      // Check both sides of the edge for the neighbour tile
      let isInner = false;
      for(const [ox,oy] of [[dx,dy],[-dx,-dy]]){
        const tx2c = Math.floor(mx + ox), ty2c = Math.floor(my + oy);
        if(tx2c < 0 || ty2c < 0 || tx2c >= WORLD_W || ty2c >= WORLD_H) continue;
        const nPid = tileProvMap[ty2c * WORLD_W + tx2c];
        if(nPid !== pid && targetSet.has(nPid)){ isInner = true; break; }
      }
      if(!isInner) outerSegs.push(x1, y1, x2, y2);
    }
  }

  if(outerSegs.length === 0){ ctx.restore(); return; }

  // Build Path2D for the outer perimeter
  const path = new Path2D();
  for(let i = 0; i < outerSegs.length; i += 4){
    const sx1 = (outerSegs[i  ] - camX) * tpx, sy1 = (outerSegs[i+1] - camY) * tpx;
    const sx2 = (outerSegs[i+2] - camX) * tpx, sy2 = (outerSegs[i+3] - camY) * tpx;
    if(sx1 < -6 && sx2 < -6) continue; if(sx1 > sw+6 && sx2 > sw+6) continue;
    if(sy1 < -6 && sy2 < -6) continue; if(sy1 > sh+6 && sy2 > sh+6) continue;
    path.moveTo(sx1, sy1); path.lineTo(sx2, sy2);
  }

  // ── 3. Draw glow passes: outer halo → mid → sharp inner ────────
  ctx.lineCap  = 'round';
  ctx.lineJoin = 'round';

  // Wide soft halo
  ctx.strokeStyle = glowCol;
  ctx.globalAlpha = 0.18;
  ctx.lineWidth   = 14;
  ctx.stroke(path);

  // Mid glow
  ctx.globalAlpha = 0.35;
  ctx.lineWidth   = 7;
  ctx.stroke(path);

  // Crisp bright edge
  ctx.globalAlpha = 0.90;
  ctx.lineWidth   = 2;
  ctx.strokeStyle = '#ffffff';
  ctx.stroke(path);

  // Coloured inner line on top
  ctx.globalAlpha = 0.70;
  ctx.lineWidth   = 1.2;
  ctx.strokeStyle = glowCol;
  ctx.stroke(path);

  ctx.restore();
}
